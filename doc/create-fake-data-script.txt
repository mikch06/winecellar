from wines.models import Wine
from django.contrib.auth.models import User
import random

user = User.objects.first()

countries = ["Frankreich", "Italien", "Spanien", "Schweiz", "Deutschland", "Österreich", "Australien"]
producers = ["Cheateau Meyney", "Chateau Laroque", "Tenuta San Guido", "Montecastro", "Delea", "Tamborini", "Tement"]

for i in range(200):
    Wine.objects.create(
        owner=user,
        winename=f"Wine {i}",
        producer=random.choice(producers),
        country=random.choice(countries),
        year=random.randint(1990, 2022),
        drinkfrom=random.randint(2020, 2028),
        drinkto=random.randint(2025, 2035),
        nmbrbottles=random.randint(0, 24),
    )

print("done")
